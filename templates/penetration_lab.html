✔ SECURED CODE (Defense-in-Depth):
@app.route('/lab/secured/profile/<int:profile_id>')
@login_required
def secured_profile(profile_id):
    profile = StudentProfile.query.get(profile_id)
    # Server-side ownership verification:
    if profile.user_id != current_user.id and current_user.role != 'admin':
        return jsonify({'error': '403 Forbidden'}), 403
    return jsonify(profile.to_dict(sensitive=True))
